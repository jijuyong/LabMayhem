cmake_minimum_required(VERSION 3.15)
project(LabMayhemProject CXX)

# Ép CMake xuất ra file bản đồ đường dẫn cho VS Code đọc hết gạch đỏ
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_CXX_STANDARD 17)

# Chỉ đường cho CMake tìm đến đúng thư mục SFML 3.0 trong ucrt64
set(SFML_DIR "C:/msys64/ucrt64/lib/cmake/SFML")

# Thêm chữ Audio vào cuối danh sách các thành phần cần tìm
find_package(SFML 3.0 COMPONENTS Graphics Window System Audio REQUIRED)

# Tạo file thực thi từ file main.cpp
add_executable(main main.cpp
    GAME.cpp)

# Thêm SFML::Audio vào cuối để liên kết âm thanh vào game
target_link_libraries(main PRIVATE SFML::Graphics SFML::Window SFML::System SFML::Audio)
